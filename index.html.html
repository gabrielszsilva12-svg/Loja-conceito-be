// ================== LOJA CONCEITO BE — Mapa de Alocação ================== // Persistência local via localStorage (sem dependências externas). const LS_COLLAB = "be_collaborators"; const LS_ALLOC = "be_allocations"; const SEAT_IDS = [ "m1-1", "m1-2", "m1-3", "m1-4", "m2-1", "m2-2", "m2-3", "m2-4", "mkt-1", "mkt-2", "mkt-3", "mkt-4", "mkt-5", "mkt-6", ]; let collaborators = loadCollaborators(); let allocations = loadAllocations(); let currentSeatId = null; // ---------- storage ---------- function loadCollaborators() { try { return JSON.parse(localStorage.getItem(LS_COLLAB)) || []; } catch (e) { return []; } } function saveCollaborators(list) { localStorage.setItem(LS_COLLAB, JSON.stringify(list)); } function loadAllocations() { try { return JSON.parse(localStorage.getItem(LS_ALLOC)) || {}; } catch (e) { return {}; } } function saveAllocations(data) { localStorage.setItem(LS_ALLOC, JSON.stringify(data)); } // ---------- helpers ---------- function pad(n) { return String(n).padStart(2, "0"); } function todayISO() { const d = new Date(); return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`; } function dateInputEl() { return document.getElementById("allocation-date"); } function currentDate() { return dateInputEl().value || todayISO(); } function escapeHtml(str) { const div = document.createElement("div"); div.textContent = str; return div.innerHTML; } // ---------- seats ---------- function renderSeats() { const date = currentDate(); const dayAlloc = allocations[date] || {}; SEAT_IDS.forEach((id) => { const btn = document.getElementById("seat-" + id); if (!btn) return; const name = dayAlloc[id]; if (name) { btn.textContent = name; btn.classList.add("occupied"); } else { btn.textContent = btn.dataset.title || "Assento"; btn.classList.remove("occupied"); } }); } function openSeatModal(seatId, title) { currentSeatId = seatId; document.getElementById("modal-seat-title").textContent = title; const select = document.getElementById("select-collaborator"); select.innerHTML = '' + collaborators .map((c) => ``) .join(""); const date = currentDate(); const current = (allocations[date] || {})[seatId] || ""; select.value = current; document.getElementById("modal-seat").classList.add("active"); } function closeSeatModal() { document.getElementById("modal-seat").classList.remove("active"); currentSeatId = null; } function saveSeatAllocation() { if (!currentSeatId) return; const name = document.getElementById("select-collaborator").value; const date = currentDate(); if (!allocations[date]) allocations[date] = {}; if (name) { allocations[date][currentSeatId] = name; } else { delete allocations[date][currentSeatId]; } saveAllocations(allocations); renderSeats(); closeSeatModal(); } function clearSeatAllocation() { if (!currentSeatId) return; const date = currentDate(); if (allocations[date]) delete allocations[date][currentSeatId]; saveAllocations(allocations); renderSeats(); closeSeatModal(); } // ---------- collaborators ---------- function openCollabModal() { renderCollabList(); document.getElementById("modal-collab").classList.add("active"); document.getElementById("new-collab-name").focus(); } function closeCollabModal() { document.getElementById("modal-collab").classList.remove("active"); } function renderCollabList() { const list = document.getElementById("collab-list"); if (collaborators.length === 0) { list.innerHTML = '
  • Nenhum colaborador cadastrado.
    • '; return; } list.innerHTML = collaborators .map( (c, i) => `
  • ${escapeHtml(c)}
    • ` ) .join(""); } function addCollaborator() { const input = document.getElementById("new-collab-name"); const name = input.value.trim(); if (!name) return; if (collaborators.includes(name)) { input.value = ""; return; } collaborators.push(name); saveCollaborators(collaborators); input.value = ""; renderCollabList(); } function removeCollaborator(index) { const removedName = collaborators[index]; collaborators.splice(index, 1); saveCollaborators(collaborators); renderCollabList(); // Remove esse colaborador de qualquer assento em que estava alocado let changed = false; Object.keys(allocations).forEach((date) => { Object.keys(allocations[date]).forEach((seatId) => { if (allocations[date][seatId] === removedName) { delete allocations[date][seatId]; changed = true; } }); }); if (changed) { saveAllocations(allocations); renderSeats(); } } // ---------- init ---------- document.addEventListener("DOMContentLoaded", () => { const dateInput = dateInputEl(); dateInput.value = todayISO(); dateInput.addEventListener("change", renderSeats); document.getElementById("new-collab-name").addEventListener("keydown", (e) => { if (e.key === "Enter") { e.preventDefault(); addCollaborator(); } }); // Fecha modais clicando fora do conteúdo document.getElementById("modal-seat").addEventListener("click", (e) => { if (e.target.id === "modal-seat") closeSeatModal(); }); document.getElementById("modal-collab").addEventListener("click", (e) => { if (e.target.id === "modal-collab") closeCollabModal(); }); renderSeats(); });
